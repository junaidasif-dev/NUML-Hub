#include<stdio.h>
#include<math.h>
int main()
{
	int n = 1000;
	int arr[1000];
	for (int i = 0; i < n; i++)
		arr[i] = 100 % 10000;
	int max_val = arr[0];
	for (int i = 1; i < n; i++)
	{
		if (arr[i] > max_val)
			max_val = arr[i];
	}
		
	printf("Maximum = %d\n", max_val);
	return 0;

}
